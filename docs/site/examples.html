Object model

// Inheritance + overriding + explicit mutation authority.
public class Entity {
    public virtual string describe() { return "Entity"; }
}
public class Player extends Entity {
    private int health = 100;
    public override string describe() { return "Player"; }
    public void damage(int amount) latch(this.health) {
        this.health -= amount;
    }
    public int getHealth() { return this.health; }
}
int main() {
    Player p = new Player();
    println(p.describe()); // Player
    p.damage(25);
    println(p.getHealth()); // 75
    return 0;
}
